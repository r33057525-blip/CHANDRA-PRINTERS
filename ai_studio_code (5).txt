  final List<CartItem> _cart = [];
  final List<CustomerOrder> _orders = [];

  List<Product> get products => _products;
  List<CartItem> get cart => _cart;
  List<CustomerOrder> get orders => _orders;

  double get cartGrandTotal => _cart.fold(0.0, (sum, item) => sum + item.calculatedTotal);

  void addToCart(CartItem item) {
    _cart.add(item);
    notifyListeners();
  }

  void removeFromCart(int index) {
    _cart.removeAt(index);
    notifyListeners();
  }

  void clearCart() {
    _cart.clear();
    notifyListeners();
  }

  CustomerOrder placeOrder({
    required String name,
    required String phone,
    required String address,
    required String paymentMethod,
    required bool isPaid,
  }) {
    final order = CustomerOrder(
      orderId: "CP-${DateTime.now().millisecondsSinceEpoch.toString().substring(7)}",
      customerName: name,
      customerPhone: phone,
      deliveryAddress: address,
      items: List.from(_cart),
      totalAmount: cartGrandTotal,
      paymentMethod: paymentMethod,
      isPaid: isPaid,
      status: OrderStatus.orderReceived,
      createdAt: DateTime.now(),
    );

    _orders.insert(0, order);
    clearCart();
    notifyListeners();
    return order;
  }

  // Admin controls
  void updateOrderStatus(String orderId, OrderStatus newStatus) {
    int index = _orders.indexWhere((o) => o.orderId == orderId);
    if (index != -1) {
      var old = _orders[index];
      _orders[index] = CustomerOrder(
        orderId: old.orderId,
        customerName: old.customerName,
        customerPhone: old.customerPhone,
        deliveryAddress: old.deliveryAddress,
        items: old.items,
        totalAmount: old.totalAmount,
        paymentMethod: old.paymentMethod,
        isPaid: old.isPaid,
        status: newStatus,
        createdAt: old.createdAt,
      );
      notifyListeners();
    }
  }

  void updateProductPrice(String productId, double newBasePrice) {
    int index = _products.indexWhere((p) => p.id == productId);
    if (index != -1) {
      var p = _products[index];
      _products[index] = Product(
        id: p.id,
        name: p.name,
        category: p.category,
        description: p.description,
        basePrice: newBasePrice,
        minQuantity: p.minQuantity,
        availableSizes: p.availableSizes,
        paperOptions: p.paperOptions,
        finishOptions: p.finishOptions,
        imageUrl: p.imageUrl,
      );
      notifyListeners();
    }
  }
